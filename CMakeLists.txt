cmake_minimum_required(VERSION 3.21)
project(darling-launcher VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Test Concurrent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0)
add_library(launcher_core src/core.cpp src/mount.cpp src/prefixbuilder.cpp)
target_include_directories(launcher_core PUBLIC src)
target_link_libraries(launcher_core PUBLIC Qt6::Core PkgConfig::PLIST)
add_executable(darling-launcher src/main.cpp src/window.cpp src/mountdialog.cpp src/appbrowser.cpp src/prefixdialog.cpp)
target_link_libraries(darling-launcher PRIVATE launcher_core Qt6::Widgets Qt6::Concurrent)
enable_testing()
add_executable(launcher-tests tests/core_test.cpp)
target_link_libraries(launcher-tests PRIVATE launcher_core Qt6::Test)
add_test(NAME launcher-tests COMMAND launcher-tests -nocrashhandler)
add_executable(launcher-mount-tests tests/mount_test.cpp)
target_link_libraries(launcher-mount-tests PRIVATE launcher_core Qt6::Test)
add_test(NAME launcher-mount-tests COMMAND launcher-mount-tests -nocrashhandler)
add_executable(launcher-prefix-tests tests/prefix_test.cpp)
target_link_libraries(launcher-prefix-tests PRIVATE launcher_core Qt6::Test)
target_compile_definitions(launcher-prefix-tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME launcher-prefix-tests COMMAND launcher-prefix-tests -nocrashhandler)
add_executable(launcher-gui-tests tests/gui_test.cpp src/window.cpp src/mountdialog.cpp src/appbrowser.cpp src/prefixdialog.cpp)
target_link_libraries(launcher-gui-tests PRIVATE launcher_core Qt6::Widgets Qt6::Concurrent Qt6::Test)
target_compile_definitions(launcher-gui-tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME launcher-gui-tests COMMAND launcher-gui-tests -nocrashhandler)
set_tests_properties(launcher-gui-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
set_tests_properties(launcher-tests launcher-mount-tests launcher-prefix-tests launcher-gui-tests PROPERTIES TIMEOUT 60)
