cmake_minimum_required(VERSION 3.21)
project(darling-launcher VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Test Concurrent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0)
add_library(launcher_core src/log.cpp src/core.cpp src/mount.cpp src/prefixbuilder.cpp src/discovery.cpp src/sources.cpp src/handoffcheck.cpp src/handoffstore.cpp src/pipelineworkspace.cpp src/stagesupervisor.cpp)
target_include_directories(launcher_core PUBLIC src)
target_link_libraries(launcher_core PUBLIC Qt6::Core PkgConfig::PLIST)
add_executable(darling-launcher src/main.cpp src/window.cpp src/mountdialog.cpp src/appbrowser.cpp src/prefixdialog.cpp src/troubleshooting.cpp src/pipeline.cpp)
target_link_libraries(darling-launcher PRIVATE launcher_core Qt6::Widgets Qt6::Concurrent)
target_compile_definitions(darling-launcher PRIVATE LAUNCHER_MOUNT_SCRIPT="${CMAKE_CURRENT_SOURCE_DIR}/tools/mount-macos.py")
enable_testing()
add_executable(launcher-tests tests/core_test.cpp)
target_link_libraries(launcher-tests PRIVATE launcher_core Qt6::Test)
add_test(NAME launcher-tests COMMAND launcher-tests -nocrashhandler)
add_executable(launcher-mount-tests tests/mount_test.cpp)
target_link_libraries(launcher-mount-tests PRIVATE launcher_core Qt6::Test)
add_test(NAME launcher-mount-tests COMMAND launcher-mount-tests -nocrashhandler)
add_executable(launcher-prefix-tests tests/prefix_test.cpp)
target_link_libraries(launcher-prefix-tests PRIVATE launcher_core Qt6::Test)
target_compile_definitions(launcher-prefix-tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME launcher-prefix-tests COMMAND launcher-prefix-tests -nocrashhandler)
add_executable(launcher-gui-tests tests/gui_test.cpp src/window.cpp src/mountdialog.cpp src/appbrowser.cpp src/prefixdialog.cpp src/troubleshooting.cpp src/pipeline.cpp)
target_link_libraries(launcher-gui-tests PRIVATE launcher_core Qt6::Widgets Qt6::Concurrent Qt6::Test)
target_compile_definitions(launcher-gui-tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" LAUNCHER_MOUNT_SCRIPT="${CMAKE_CURRENT_SOURCE_DIR}/tools/mount-macos.py")
add_test(NAME launcher-gui-tests COMMAND launcher-gui-tests -nocrashhandler)
set_tests_properties(launcher-gui-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(launcher-troubleshooting-tests tests/troubleshooting_test.cpp src/troubleshooting.cpp src/pipeline.cpp)
target_link_libraries(launcher-troubleshooting-tests PRIVATE launcher_core Qt6::Widgets Qt6::Concurrent Qt6::Test)
add_test(NAME launcher-troubleshooting-tests COMMAND launcher-troubleshooting-tests -nocrashhandler)
set_tests_properties(launcher-troubleshooting-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
set_tests_properties(launcher-tests launcher-mount-tests launcher-prefix-tests launcher-gui-tests PROPERTIES TIMEOUT 240)
add_executable(launcher-handoff-tests tests/handoff_check_test.cpp)
target_link_libraries(launcher-handoff-tests PRIVATE launcher_core Qt6::Test)
target_compile_definitions(launcher-handoff-tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME launcher-handoff-tests COMMAND launcher-handoff-tests -nocrashhandler)
add_executable(launcher-handoff-store-tests tests/handoff_store_test.cpp)
target_link_libraries(launcher-handoff-store-tests PRIVATE launcher_core Qt6::Test)
target_compile_definitions(launcher-handoff-store-tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME launcher-handoff-store-tests COMMAND launcher-handoff-store-tests -nocrashhandler)
add_executable(launcher-pipeline-workspace-tests tests/pipeline_workspace_test.cpp)
target_link_libraries(launcher-pipeline-workspace-tests PRIVATE launcher_core Qt6::Test)
add_test(NAME launcher-pipeline-workspace-tests COMMAND launcher-pipeline-workspace-tests -nocrashhandler)
add_executable(launcher-stage-supervisor-tests tests/stage_supervisor_test.cpp)
target_link_libraries(launcher-stage-supervisor-tests PRIVATE launcher_core Qt6::Test)
add_test(NAME launcher-stage-supervisor-tests COMMAND launcher-stage-supervisor-tests -nocrashhandler)
add_executable(launcher-pipeline-tests tests/pipeline_test.cpp src/pipeline.cpp src/troubleshooting.cpp)
target_link_libraries(launcher-pipeline-tests PRIVATE launcher_core Qt6::Widgets Qt6::Concurrent Qt6::Test)
target_compile_definitions(launcher-pipeline-tests PRIVATE TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME launcher-pipeline-tests COMMAND launcher-pipeline-tests -nocrashhandler)
set_tests_properties(launcher-pipeline-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;PYTHONDONTWRITEBYTECODE=1")
set_tests_properties(launcher-handoff-tests launcher-handoff-store-tests launcher-pipeline-workspace-tests launcher-pipeline-tests PROPERTIES TIMEOUT 240)

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME launcher-mount-script-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mount_script_test.py)
    set_tests_properties(launcher-mount-script-tests PROPERTIES TIMEOUT 30)
endif()

qt_add_resources(darling-launcher runtime_builder PREFIX "/launcher" FILES tools/all-vibedarling-pr-prefix.py)
qt_add_resources(launcher-gui-tests test_runtime_builder PREFIX "/launcher" FILES tools/all-vibedarling-pr-prefix.py)
