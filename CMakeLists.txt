cmake_minimum_required(VERSION 3.21)
project(darling-launcher VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Test Concurrent)
find_package(PkgConfig REQUIRED)
include(GNUInstallDirs)
# Root-owned mount helper run through pkexec; -DLAUNCHER_MOUNT_SCRIPT=... overrides, and install() follows it.
if(NOT LAUNCHER_MOUNT_SCRIPT)
    set(LAUNCHER_MOUNT_SCRIPT "${CMAKE_INSTALL_FULL_LIBEXECDIR}/darling-launcher/mount-macos.py")
endif()
get_filename_component(LAUNCHER_MOUNT_SCRIPT_DIR "${LAUNCHER_MOUNT_SCRIPT}" DIRECTORY)
get_filename_component(LAUNCHER_MOUNT_SCRIPT_NAME "${LAUNCHER_MOUNT_SCRIPT}" NAME)
pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0)
add_library(launcher_core src/log.cpp src/core.cpp src/mount.cpp src/prefixbuilder.cpp src/discovery.cpp src/sources.cpp src/releases.cpp src/handoffcheck.cpp src/handoffstore.cpp src/pipelineworkspace.cpp src/stagesupervisor.cpp src/libraries.cpp src/macho.cpp src/imagesource.cpp src/prefixstate.cpp src/safefs.cpp src/libraryimport.cpp src/libraryops.cpp src/librarystatus.cpp)
target_include_directories(launcher_core PUBLIC src)
target_link_libraries(launcher_core PUBLIC Qt6::Core PkgConfig::PLIST)
add_executable(darling-launcher src/main.cpp src/window.cpp src/mountdialog.cpp src/appbrowser.cpp src/prefixdialog.cpp src/troubleshooting.cpp src/pipeline.cpp)
target_link_libraries(darling-launcher PRIVATE launcher_core Qt6::Widgets Qt6::Concurrent)
target_compile_definitions(darling-launcher PRIVATE LAUNCHER_MOUNT_SCRIPT="${LAUNCHER_MOUNT_SCRIPT}")
install(PROGRAMS tools/mount-macos.py DESTINATION "${LAUNCHER_MOUNT_SCRIPT_DIR}" RENAME "${LAUNCHER_MOUNT_SCRIPT_NAME}")
