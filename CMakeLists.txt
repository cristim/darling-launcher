cmake_minimum_required(VERSION 3.21)
project(darling-launcher VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Test Concurrent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0)
add_library(launcher_core src/log.cpp src/core.cpp src/mount.cpp src/prefixbuilder.cpp src/discovery.cpp src/sources.cpp src/libraries.cpp src/macho.cpp src/imagesource.cpp src/prefixstate.cpp src/safefs.cpp src/libraryimport.cpp src/libraryops.cpp src/librarystatus.cpp src/libscli.cpp src/dyldcacheheader.cpp src/dyldcachestore.cpp src/dyldcachecopy.cpp src/dyldcacheconfig.cpp src/launchenv.cpp)
target_include_directories(launcher_core PUBLIC src)
target_link_libraries(launcher_core PUBLIC Qt6::Core PkgConfig::PLIST)
add_executable(darling-launcher src/main.cpp src/window.cpp src/mountdialog.cpp src/appbrowser.cpp src/prefixdialog.cpp src/troubleshooting.cpp)
target_link_libraries(darling-launcher PRIVATE launcher_core Qt6::Widgets Qt6::Concurrent)
target_compile_definitions(darling-launcher PRIVATE LAUNCHER_MOUNT_SCRIPT="${CMAKE_CURRENT_SOURCE_DIR}/tools/mount-macos.py")
enable_testing()
add_executable(launcher-tests tests/core_test.cpp)
target_link_libraries(launcher-tests PRIVATE launcher_core Qt6::Test)

target_link_libraries(launcher-libraries-tests PRIVATE launcher_core Qt6::Test)

target_link_libraries(launcher-status-tests PRIVATE launcher_core Qt6::Test)
add_test(NAME launcher-status-tests COMMAND launcher-status-tests -nocrashhandler)
